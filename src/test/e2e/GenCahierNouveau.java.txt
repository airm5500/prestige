/**
 * Ajoute au cahier de recette existant les feuilles des fonctionnalites livrees sur la
 * fiche article : liste, apercu, fiche detail, lots, parametrage des actions, connexion
 * et montee en charge. Le format (colonnes, styles, synthese) est celui des feuilles
 * deja presentes, pour que le document reste homogene.
 */
public class GenCahierNouveau {

    static Workbook wb;
    static CellStyle titre, sousTitre, entete, section, cell, cellC, saisie, synthLabel;

    public static void main(String[] args) throws Exception {
        String fichier = args.length > 0 ? args[0] : "Cahier_de_recette_complet.xlsx";
        try (InputStream in = new FileInputStream(fichier)) { wb = new XSSFWorkbook(in); }
        // Regeneration : on repart d'un document sans les feuilles produites ici.
        for (String nom : new String[]{"Fiche article - Liste", "Fiche article - Apercu",
                "Fiche article - Detail", "Lots et parametrage", "Connexion et charge"}) {
            int i = wb.getSheetIndex(nom);
            if (i >= 0) { wb.removeSheetAt(i); }
        }
        styles();
        feuilleListe();
        feuilleApercu();
        feuilleDetail();
        feuilleLotsParametrage();
        feuilleConnexionCharge();
        try (FileOutputStream fos = new FileOutputStream(fichier)) { wb.write(fos); }
        wb.close();
        System.out.println("Feuilles ajoutees dans : " + fichier);
    }

    // ------------------------------------------------------------------ OUTILS DE MISE EN FORME
    static void largeurs(Sheet sh) {
        sh.setColumnWidth(0, 1400);
        sh.setColumnWidth(1, 3200);
        sh.setColumnWidth(2, 8200);
        sh.setColumnWidth(3, 7200);
        sh.setColumnWidth(4, 11000);
        sh.setColumnWidth(5, 12500);
        sh.setColumnWidth(6, 8500);
        sh.setColumnWidth(7, 3200);
        sh.setColumnWidth(8, 5200);
    }

    static void titreLigne(Sheet sh, int[] r, String t) {
        Row row = sh.createRow(r[0]); row.setHeightInPoints(26);
        Cell c = row.createCell(0); c.setCellValue(t); c.setCellStyle(titre);
        sh.addMergedRegion(new CellRangeAddress(r[0], r[0], 0, 8)); r[0]++;
    }

    static void sousLigne(Sheet sh, int[] r, String t) {
        Row row = sh.createRow(r[0]); row.setHeightInPoints(54);
        Cell c = row.createCell(0); c.setCellValue(t); c.setCellStyle(sousTitre);
        sh.addMergedRegion(new CellRangeAddress(r[0], r[0], 0, 8)); r[0]++;
        r[0]++;
    }

    static void entetes(Sheet sh, int[] r) {
        Row row = sh.createRow(r[0]); row.setHeightInPoints(28);
        String[] h = {"N", "Section", "Objet du controle", "Pre-requis", "Mode operatoire", "Resultat attendu",
                "Resultat obtenu", "Statut", "Testeur / Date"};
        for (int i = 0; i < h.length; i++) { Cell c = row.createCell(i); c.setCellValue(h[i]); c.setCellStyle(entete); }
        r[0]++;
        sh.createFreezePane(0, r[0]);
    }

    static void remplir(Sheet sh, int[] r, String[][] d) {
        for (String[] ligne : d) {
            if (ligne[0].equals("S")) {
                Row row = sh.createRow(r[0]); row.setHeightInPoints(18);
                Cell c = row.createCell(0); c.setCellValue(ligne[1]); c.setCellStyle(section);
                for (int i = 1; i <= 8; i++) { row.createCell(i).setCellStyle(section); }
                sh.addMergedRegion(new CellRangeAddress(r[0], r[0], 0, 8)); r[0]++;
            } else {
                Row row = sh.createRow(r[0]);
                int maxLen = 0;
                for (int i = 2; i < ligne.length; i++) { maxLen = Math.max(maxLen, ligne[i].length()); }
                row.setHeightInPoints(Math.max(30, (float) (Math.ceil(maxLen / 40.0) * 13)));
                Cell c0 = row.createCell(0); c0.setCellValue(ligne[0]); c0.setCellStyle(cellC);
                for (int i = 1; i <= 5; i++) {
                    Cell c = row.createCell(i);
                    c.setCellValue(i < ligne.length ? ligne[i] : "");
                    c.setCellStyle(cell);
                }
                for (int i = 6; i <= 8; i++) { row.createCell(i).setCellStyle(saisie); }
                r[0]++;
            }
        }
    }

    static void synthese(Sheet sh, int[] r, int premiereLigne) {
        r[0]++;
        String plage = "$A$" + premiereLigne + ":$A$" + (r[0] - 1);
        String plageStatut = "$H$" + premiereLigne + ":$H$" + (r[0] - 1);
        ligneSynth(sh, r, "SYNTHESE", null);
        ligneSynth(sh, r, "Controles au total", "COUNTA(" + plage + ")-COUNTIF(" + plage + ",\"\")");
        ligneSynth(sh, r, "Conformes (OK)", "COUNTIF(" + plageStatut + ",\"OK\")");
        ligneSynth(sh, r, "Non conformes (KO)", "COUNTIF(" + plageStatut + ",\"KO\")");
        ligneSynth(sh, r, "Sans objet (N/A)", "COUNTIF(" + plageStatut + ",\"N/A\")");
        r[0]++;
        Row row = sh.createRow(r[0]); row.setHeightInPoints(22);
        Cell c = row.createCell(0);
        c.setCellValue("Decision de recette :   [ ] Acceptee     [ ] Acceptee avec reserves     [ ] Refusee"
                + "        Responsable : __________________     Date : ____/____/______");
        c.setCellStyle(sousTitre);
        sh.addMergedRegion(new CellRangeAddress(r[0], r[0], 0, 8)); r[0]++;
    }

    static void ligneSynth(Sheet sh, int[] r, String label, String formule) {
        Row row = sh.createRow(r[0]);
        Cell c0 = row.createCell(0); c0.setCellValue(label); c0.setCellStyle(synthLabel);
        sh.addMergedRegion(new CellRangeAddress(r[0], r[0], 0, 4));
        for (int i = 1; i <= 4; i++) { row.createCell(i).setCellStyle(synthLabel); }
        Cell c5 = row.createCell(5);
        if (formule != null) { c5.setCellFormula(formule); }
        c5.setCellStyle(synthLabel);
        for (int i = 6; i <= 8; i++) { row.createCell(i).setCellStyle(synthLabel); }
        r[0]++;
    }

    static void styles() {
        Font fT = wb.createFont(); fT.setBold(true); fT.setFontHeightInPoints((short) 14);
        titre = wb.createCellStyle(); titre.setFont(fT); titre.setAlignment(HorizontalAlignment.CENTER);
        titre.setVerticalAlignment(VerticalAlignment.CENTER);
        Font fS = wb.createFont(); fS.setItalic(true); fS.setFontHeightInPoints((short) 9);
        sousTitre = wb.createCellStyle(); sousTitre.setFont(fS); sousTitre.setWrapText(true);
        sousTitre.setVerticalAlignment(VerticalAlignment.CENTER);
        Font fE = wb.createFont(); fE.setBold(true); fE.setColor(IndexedColors.WHITE.getIndex());
        entete = wb.createCellStyle(); entete.setFont(fE); entete.setFillForegroundColor(IndexedColors.DARK_BLUE.getIndex());
        entete.setFillPattern(FillPatternType.SOLID_FOREGROUND); entete.setAlignment(HorizontalAlignment.CENTER);
        entete.setVerticalAlignment(VerticalAlignment.CENTER); entete.setWrapText(true); bord(entete);
        Font fSec = wb.createFont(); fSec.setBold(true); fSec.setColor(IndexedColors.WHITE.getIndex());
        section = wb.createCellStyle(); section.setFont(fSec);
        section.setFillForegroundColor(IndexedColors.GREY_50_PERCENT.getIndex());
        section.setFillPattern(FillPatternType.SOLID_FOREGROUND);
        section.setVerticalAlignment(VerticalAlignment.CENTER); bord(section);
        cell = wb.createCellStyle(); cell.setWrapText(true); cell.setVerticalAlignment(VerticalAlignment.TOP); bord(cell);
        cellC = wb.createCellStyle(); cellC.setAlignment(HorizontalAlignment.CENTER);
        cellC.setVerticalAlignment(VerticalAlignment.TOP); bord(cellC);
        saisie = wb.createCellStyle(); saisie.setFillForegroundColor(IndexedColors.LEMON_CHIFFON.getIndex());
        saisie.setFillPattern(FillPatternType.SOLID_FOREGROUND); saisie.setWrapText(true);
        saisie.setVerticalAlignment(VerticalAlignment.TOP); bord(saisie);
        Font fSl = wb.createFont(); fSl.setBold(true);
        synthLabel = wb.createCellStyle(); synthLabel.setFont(fSl); bord(synthLabel);
        synthLabel.setVerticalAlignment(VerticalAlignment.CENTER);
    }

    static void bord(CellStyle s) {
        s.setBorderBottom(BorderStyle.THIN); s.setBorderTop(BorderStyle.THIN);
        s.setBorderLeft(BorderStyle.THIN); s.setBorderRight(BorderStyle.THIN);
    }
